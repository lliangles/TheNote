## 📌 1. Python 模組 (Modules) 與標準函式庫

```python
# ==============================================================================
# 範例程式 6-1：常用標準庫 math, random, datetime 實戰
# ==============================================================================
import math
import random
from datetime import datetime, timedelta

# 1. math 數學函式庫
print(f"圓周率 pi: {math.pi:.5f}")
print(f"e 的自然對數 ln(e^3): {math.log(math.e ** 3):.1f}")
print(f"無條件進位 ceil(4.2): {math.ceil(4.2)}")

# 2. random 隨機函式庫（常用於機器學習資料抽樣與打亂）
random.seed(42)  # 固定隨機種子，確保實驗可重現 (Reproducibility)
dice_roll = random.randint(1, 6)             # 隨機整數 [1, 6]
samples = random.sample(range(1, 100), 5)     # 從母體隨機抽取 5 個不重複樣本
print(f"骰子點數: {dice_roll}, 隨機樣本: {samples}")

# 3. datetime 時間與日期處理
now = datetime.now()
formatted_time = now.strftime("%Y-%m-%d %H:%M:%S")
one_week_later = now + timedelta(days=7)
print(f"當前系統時間: {formatted_time}")
print(f"七天後截止日: {one_week_later.strftime('%Y-%m-%d')}")
```

## 📌 2. 檔案讀寫與 Context Manager (`with` 語法)

在讀寫檔案時，**務必使用 `with open(...)` 上下文管理器**：
- 保證即使在檔案讀寫過程發生崩潰或異常，作業系統檔案句柄 (File Handle) **必會自動關閉**，防止記憶體洩漏與檔案鎖死。
- 中文編碼處理：**務必顯式指定 `encoding="utf-8"`**！

```python
# ==============================================================================
# 範例程式 6-2：純文字、CSV 與 JSON 檔案讀寫實務
# ==============================================================================
import json
import csv
import os

os.makedirs("data_temp", exist_ok=True)
json_file = "data_temp/model_metrics.json"
csv_file = "data_temp/students.csv"

# 1. JSON 檔案寫入與讀取 (機器學習紀錄常用)
metrics_data = {
    "model": "RandomForest",
    "accuracy": 0.945,
    "epochs": 50,
    "hyperparameters": {"n_trees": 100, "max_depth": 10}
}

# 寫入 JSON (ensure_ascii=False 支援中文字符，indent=4 自動美化縮排)
with open(json_file, "w", encoding="utf-8") as f:
    json.dump(metrics_data, f, ensure_ascii=False, indent=4)

# 讀取 JSON
with open(json_file, "r", encoding="utf-8") as f:
    loaded_metrics = json.load(f)
print(f"讀出 JSON 模型準確度: {loaded_metrics['accuracy']}")

# 2. CSV 結構化資料寫入與讀取
students_data = [
    ["學號", "姓名", "AI分數"],
    ["A01", "張小明", "95"],
    ["A02", "李小華", "88"]
]

# 寫入 CSV (newline='' 防止 Windows 系統產生多餘空行)
with open(csv_file, "w", encoding="utf-8", newline="") as f:
    writer = csv.writer(f)
    writer.writerows(students_data)

# 讀取 CSV
with open(csv_file, "r", encoding="utf-8") as f:
    reader = csv.reader(f)
    print("\n--- 讀取 CSV 內容 ---")
    for row in reader:
        print(f"  {row}")

# 清理測試檔案
os.remove(json_file)
os.remove(csv_file)
os.rmdir("data_temp")
```

## 📌 3. 結構化例外與異常處理 (try-except-else-finally)

```python
# ==============================================================================
# 範例程式 6-3：全功能例外處理與自訂例外 (Custom Exception)
# ==============================================================================

class ScoreOutOfRangeException(Exception):
    """自訂成績範圍異常"""
    pass

def validate_and_process_score(raw_input: str) -> float:
    """
    驗證並計算成績
    展示 try-except-else-finally 完整防護架構
    """
    try:
        score = float(raw_input)
        if not (0 <= score <= 100):
            # 主動丟出自訂例外
            raise ScoreOutOfRangeException(f"輸入數值 {score} 不在合理區間 [0, 100]！")

        adjustment = 100 / score  # 若輸入 0 會引發 ZeroDivisionError

    except ValueError:
        print("❌ [錯誤捕獲] 輸入非合法數字格式！")
        return -1.0
    except ZeroDivisionError:
        print("❌ [錯誤捕獲] 成績為 0 分，無法進行除法加權！")
        return 0.0
    except ScoreOutOfRangeException as err:
        print(f"⚠️ [自訂例外捕獲] {err}")
        return -1.0
    else:
        # 只有在 try 區塊「完全沒有拋出任何例外」時才會執行
        print(f"✅ 輸入驗證通過，加權倍數為: {adjustment:.2f}")
        return score
    finally:
        # 無論有無發生例外，最後「必定保證執行」（通常用於釋放資源）
        print("   -> 完成單次檢核事務 (Transaction Closed)\n")

# 測試各類邊界異常狀況
print("--- 測試 1: 合法輸入 80 ---")
validate_and_process_score("80")

print("--- 測試 2: 字串非數字 'abc' ---")
validate_and_process_score("abc")

print("--- 測試 3: 數值為 0 ---")
validate_and_process_score("0")

print("--- 測試 4: 超出範圍 150 ---")
validate_and_process_score("150")
```
