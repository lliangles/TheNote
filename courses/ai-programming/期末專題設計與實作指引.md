## 📌 2. 期末專題端到端完整範本架構 (End-to-End Pipeline)

```python
# ==============================================================================
# 期末專題示範範本：金融大數據擷取、Pandas 清洗與 Matplotlib 視覺化全流程
# ==============================================================================
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

plt.rcParams['font.sans-serif'] = ['Microsoft JhengHei']
plt.rcParams['axes.unicode_minus'] = False

def run_project_pipeline():
    print("=== 步驟 1: 資料收集與模擬 (Data Ingestion) ===")
    dates = pd.date_range("2024-01-01", periods=120)
    prices = 100 * np.cumprod(1 + np.random.normal(0.001, 0.02, 120))
    df = pd.DataFrame({"Date": dates, "Price": prices}).set_index("Date")

    print("=== 步驟 2: 資料清洗與特徵工程 (Feature Engineering) ===")
    df["MA_10"] = df["Price"].rolling(10).mean()
    df["MA_30"] = df["Price"].rolling(30).mean()
    df["Daily_Return"] = df["Price"].pct_change()

    print("=== 步驟 3: 策略信號判定與回測 (Backtesting) ===")
    df["Signal"] = np.where(df["MA_10"] > df["MA_30"], 1, 0)
    df["Strategy_Return"] = df["Signal"].shift(1) * df["Daily_Return"]
    df["Cum_Benchmark"] = (1 + df["Daily_Return"]).cumprod()
    df["Cum_Strategy"] = (1 + df["Strategy_Return"]).cumprod()

    print("=== 步驟 4: 圖表視覺化輸出 (Visualization) ===")
    fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(12, 8), sharex=True)

    # 上圖: 價格與均線
    ax1.plot(df.index, df["Price"], label="資產價格", color="black", alpha=0.7)
    ax1.plot(df.index, df["MA_10"], label="10日短期均線", color="orange")
    ax1.plot(df.index, df["MA_30"], label="30日中期均線", color="blue")
    ax1.set_title("期末專題成果展示：資產走勢與雙均線策略圖", fontsize=14)
    ax1.legend()
    ax1.grid(True, linestyle="--")

    # 下圖: 累積回報對比
    ax2.plot(df.index, df["Cum_Benchmark"], label="基準報酬 (Benchmark)", color="gray", linestyle="--")
    ax2.plot(df.index, df["Cum_Strategy"], label="量化策略報酬 (Strategy)", color="red", linewidth=2)
    ax2.set_title("策略累積報酬率對比", fontsize=14)
    ax2.legend()
    ax2.grid(True, linestyle="--")

    plt.tight_layout()
    plt.savefig("project_backtest_result.png", dpi=150)
    print("🎉 專題圖表已產出至 project_backtest_result.png！")

if __name__ == "__main__":
    run_project_pipeline()
```
