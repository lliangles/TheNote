## 📌 1. 網路爬蟲基本運作原理與 HTTP 協定

1. **Client 送出 HTTP Request (請求)**：
   - 請求方法：`GET`（讀取資料）、`POST`（提交表單/帳密登入）。
   - 請求標頭 (Headers)：`User-Agent`（偽裝瀏覽器身分）、`Cookie`（記錄登入狀態）。
2. **Server 回應 HTTP Response (回應)**：
   - 狀態碼：`200 OK`（成功）、`403 Forbidden`（被反爬擋下）、`404 Not Found`、`500 Server Error`。
   - 回傳主體：HTML 網頁原始碼、JSON 結構化資料。

## 📌 2. 靜態網頁解析：Requests + BeautifulSoup

```python
# ==============================================================================
# 範例程式 10-1：PTT 看板文章標題與推文數爬取實作
# 說明：展示偽裝 User-Agent、Cookie 滿 18 歲確認，以及 CSS Selector 解析
# ==============================================================================
import requests
from bs4 import BeautifulSoup

url = "https://www.ptt.cc/bbs/Gossiping/index.html"

# 1. 關鍵偽裝：設定 Headers 模擬 Chrome 瀏覽器
headers = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36"
}

# 2. 繞過年齡確認彈窗：在 Cookie 中注入 over18=1
cookies = {"over18": "1"}

try:
    response = requests.get(url, headers=headers, cookies=cookies, timeout=10)
    response.encoding = "utf-8"  # 指定編碼避免繁體中文亂碼

    if response.status_code == 200:
        # 使用 BeautifulSoup 與 lxml/html.parser 解析 HTML
        soup = BeautifulSoup(response.text, "html.parser")

        # 使用 CSS 選擇器選取所有文章列 (.r-ent)
        articles = soup.select("div.r-ent")
        print(f"成功擷取 {len(articles)} 篇熱門文章：\n")

        for art in articles[:10]:
            title_tag = art.select_one("div.title a")
            push_tag = art.select_one("div.nrec span")
            author_tag = art.select_one("div.meta div.author")

            title = title_tag.text.strip() if title_tag else "[本文已刪除]"
            push_count = push_tag.text.strip() if push_tag else "0"
            author = author_tag.text.strip() if author_tag else "N/A"

            print(f"推文: {push_count:>3} | 作者: {author:<15} | 標題: {title}")
    else:
        print(f"請求失敗，HTTP 狀態碼: {response.status_code}")

except Exception as e:
    print(f"爬蟲連線異常: {e}")
```

## 📌 3. 批次圖片大量下載實作 (對應課堂 Ch07)

```python
# ==============================================================================
# 範例程式 10-2：批次爬取網路圖片並串流存入硬碟
# 說明：對應溫敏淦教授 F1700_ch07 課堂主題
# ==============================================================================
import requests
import os

def download_images_batch(image_urls: list[str], save_directory: str = "downloaded_imgs"):
    """
    大量圖片批次下載器（採 stream 模式防爆記憶體）
    """
    os.makedirs(save_directory, exist_ok=True)
    headers = {"User-Agent": "Mozilla/5.0"}

    for idx, img_url in enumerate(image_urls, start=1):
        try:
            # stream=True 採資料流分塊下載，適合大圖或大容量檔案
            res = requests.get(img_url, headers=headers, stream=True, timeout=15)
            if res.status_code == 200:
                file_extension = img_url.split(".")[-1].split("?")[0]
                if file_extension.lower() not in ["jpg", "jpeg", "png", "webp"]:
                    file_extension = "jpg"

                file_path = os.path.join(save_directory, f"image_{idx:03d}.{file_extension}")
                with open(file_path, "wb") as f:
                    for chunk in res.iter_content(chunk_size=1024 * 32):
                        f.write(chunk)
                print(f"  [下載成功] {file_path}")
        except Exception as e:
            print(f"  [下載失敗] {img_url}，原因: {e}")

# 測試呼叫示範
test_urls = [
    "https://httpbin.org/image/png",
    "https://httpbin.org/image/jpeg"
]
download_images_batch(test_urls)
```

## 📌 4. 動態網頁渲染：Selenium 4 + ChromeDriver

對於使用 Vue, React 透過 AJAX 動態渲染的現代網頁，傳統 `requests` 無法取得由 JavaScript 動態載入的內容，必須透過**自動化瀏覽器 (Headless Browser)** 抓取：

```python
# ==============================================================================
# 範例程式 10-3：Selenium 4 無頭瀏覽器 (Headless) 爬蟲範本
# 說明：對應教材中 chromedriver-win64 配置
# ==============================================================================
from selenium import webdriver
from selenium.webdriver.chrome.service import Service
from selenium.webdriver.chrome.options import Options
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC

def run_headless_crawler(target_url: str):
    """
    使用 Selenium 驅動 Chrome 進行動態 DOM 元素爬取
    """
    # 1. 配置 Chrome 瀏覽器選項
    chrome_options = Options()
    chrome_options.add_argument("--headless")              # 無介面背景執行 (節省資源)
    chrome_options.add_argument("--disable-gpu")
    chrome_options.add_argument("--no-sandbox")
    chrome_options.add_argument("window-size=1920x1080")
    chrome_options.add_argument("user-agent=Mozilla/5.0 (Windows NT 10.0; Win64; x64)")

    # 2. 啟動瀏覽器驅動器 (Selenium 4 支援自動管理驅動程式)
    driver = webdriver.Chrome(options=chrome_options)

    try:
        driver.get(target_url)

        # 3. 顯式等待 (Explicit Wait)：最多等待 10 秒直到特定元素渲染完成
        wait = WebDriverWait(driver, 10)
        body_element = wait.until(EC.presence_of_element_located((By.TAG_NAME, "body")))

        page_title = driver.title
        print(f"網頁標題: {page_title}")
        print(f"頁面渲染 HTML 字數: {len(driver.page_source)} 字")

    finally:
        # 務必釋放資源關閉背景 Chrome 行程
        driver.quit()

# 測試範例（可直接執行）
run_headless_crawler("https://www.google.com")
```
